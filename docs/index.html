Quick Start

npm install audio-tracker
import AudioTracker from 'audio-tracker';

const tracker = new AudioTracker('/audio.mp3', {
  volume: 80,
  mediaSession: { title: 'My Song', artist: 'Artist' }
});

tracker.init({
  onPlay: () => console.log('Playing'),
  onTimeUpdate: (time) => console.log(time)
});

tracker.play();
tracker.forward(10);
tracker.backward(10);
tracker.setVolume(80);
odede>function Player() {
  const ref = useRef(null);
  const [playing, setPlaying] = useState(false);

  useEffect(() => {
    ref.current = new AudioTracker('/audio.mp3');
    ref.current.init({
      onPlay: () => setPlaying(true),
      onPause: () => setPlaying(false)
    });
    return () => ref.current?.destroy();
  }, []);

  return (
    <button onClick={() => 
      ref.current.isPlaying() 
        ? ref.current.pause() 
        : ref.current.play()
    }>
      {playing ? 'Pause' : 'Play'}
    </button>
  );
}
