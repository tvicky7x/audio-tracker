Implementation

npm install audio-tracker
import AudioTracker from 'audio-tracker';

const tracker = new AudioTracker('/audio.mp3', {
  preload: 'metadata',
  loop: false,
  volume: 80,
  mediaSession: {
    title: 'My Song',
    artist: 'Artist Name'
  }
});

tracker.init({
  onPlay: () => console.log('Playing'),
  onPause: () => console.log('Paused'),
  onTimeUpdate: (time) => console.log(time),
  onBufferChangePercentage: (pct) => {
    console.log(`Buffered: ${pct}%`);
  }
});

// Control playback
tracker.play();
tracker.pause();
tracker.seekTo(30);
tracker.setVolume(80);
tracker.setPlaybackRate(1.5);
import { useEffect, useRef, useState } from 'react';
import AudioTracker from 'audio-tracker';

function Player() {
  const tracker = useRef(null);
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(0);

  useEffect(() => {
    tracker.current = new AudioTracker('/audio.mp3');
    
    tracker.current.init({
      onPlay: () => setIsPlaying(true),
      onPause: () => setIsPlaying(false),
      onTimeUpdate: (time) => setCurrentTime(time)
    });

    return () => tracker.current?.destroy();
  }, []);

  return (
    <button onClick={() => {
      isPlaying 
        ? tracker.current.pause() 
        : tracker.current.play();
    }}>
      {isPlaying ? 'Pause' : 'Play'}
    </button>
  );
}
